cmake_minimum_required(VERSION 3.24)
project(MyCut VERSION 0.11.0 LANGUAGES CXX)
if(MSVC)
    add_compile_options(/W4 /permissive- /utf-8)
    get_filename_component(MYCUT_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
    # A Russian-only MSVC installation cannot honor VSLANG=1033. Ninja needs its actual prefix.
    if(EXISTS "${MYCUT_COMPILER_DIR}/1049/clui.dll" AND NOT EXISTS "${MYCUT_COMPILER_DIR}/1033/clui.dll")
        set(CMAKE_CL_SHOWINCLUDES_PREFIX "Примечание: включение файла:")
    endif()
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_library(MyCutCore STATIC
    engine/src/Project.cpp engine/src/Serialization.cpp engine/src/Session.cpp engine/src/Interaction.cpp engine/src/Timeline.cpp engine/src/Nodes.cpp
    engine/src/Parameters.cpp engine/src/LegacyParameterAdapter.cpp)
target_include_directories(MyCutCore PUBLIC engine/include)
if(MSVC)
    target_compile_options(MyCutCore PRIVATE /W4 /permissive- /utf-8)
endif()
add_executable(MyCutCoreTests tests/CoreTests.cpp)
target_link_libraries(MyCutCoreTests PRIVATE MyCutCore)
add_test(NAME Core COMMAND MyCutCoreTests)
add_executable(MyCutInteractionTests tests/InteractionTests.cpp)
target_link_libraries(MyCutInteractionTests PRIVATE MyCutCore)
add_test(NAME Interaction COMMAND MyCutInteractionTests)
add_executable(MyCutTimelineTests tests/TimelineTests.cpp)
target_link_libraries(MyCutTimelineTests PRIVATE MyCutCore)
add_test(NAME Timeline COMMAND MyCutTimelineTests)
add_executable(MyCutNodeTests tests/NodeTests.cpp)
target_link_libraries(MyCutNodeTests PRIVATE MyCutCore)
add_test(NAME Nodes COMMAND MyCutNodeTests)
add_executable(MyCutParameterTests tests/ParameterTests.cpp)
target_link_libraries(MyCutParameterTests PRIVATE MyCutCore)
add_test(NAME Parameters COMMAND MyCutParameterTests)
add_executable(MyCutParameterPerformanceTests tests/ParameterPerformanceTests.cpp)
target_link_libraries(MyCutParameterPerformanceTests PRIVATE MyCutCore)
add_test(NAME ParameterPerformance COMMAND MyCutParameterPerformanceTests)
if(WIN32)
    add_library(MyCutGraphics STATIC native/src/D3D11Renderer.cpp native/src/Generators.cpp native/src/Media.cpp)
    target_include_directories(MyCutGraphics PUBLIC native/include)
    target_link_libraries(MyCutGraphics PUBLIC MyCutCore d3d11 d3d9 dxgi d3dcompiler windowscodecs ole32 d2d1 dwrite mfplat mfreadwrite mfuuid propsys)
    target_compile_definitions(MyCutGraphics PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_library(MyCutNative SHARED native/src/Bridge.cpp)
    target_link_libraries(MyCutNative PRIVATE MyCutGraphics)
    target_include_directories(MyCutNative PUBLIC native/include)
    add_executable(MyCutBridgeTests tests/BridgeTests.cpp)
    target_link_libraries(MyCutBridgeTests PRIVATE MyCutNative)
    add_test(NAME Bridge COMMAND MyCutBridgeTests)
    add_executable(MyCutRenderTests tests/RenderTests.cpp)
    target_link_libraries(MyCutRenderTests PRIVATE MyCutGraphics)
    add_test(NAME Render COMMAND MyCutRenderTests "${CMAKE_BINARY_DIR}/render-test.png")
    add_executable(MyCutMediaTests tests/MediaTests.cpp)
    target_link_libraries(MyCutMediaTests PRIVATE MyCutGraphics)
    add_test(NAME Media COMMAND MyCutMediaTests "${CMAKE_BINARY_DIR}")
    foreach(target MyCutGraphics MyCutNative MyCutRenderTests)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    endforeach()
endif()
